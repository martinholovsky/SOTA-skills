    updated_at  TEXT NOT NULL DEFAULT (strftime('%Y-%m-%dT%H:%M:%fZ', 'now'))
);

CREATE INDEX IF NOT EXISTS idx_reports_tenant_owner
    ON reports (tenant_id, owner_id, created_at DESC);

CREATE INDEX IF NOT EXISTS idx_reports_tenant_created
    ON reports (tenant_id, created_at DESC);

CREATE TABLE IF NOT EXISTS attachments (
    id              TEXT PRIMARY KEY,      -- UUID v4

    storage_key     TEXT NOT NULL UNIQUE,  -- server-generated UUID path segment
    size_bytes      INTEGER NOT NULL,
    content_type    TEXT NOT NULL,

import sqlite3
import os
import pathlib
import logging

logger = logging.getLogger(__name__)

_SCHEMA = pathlib.Path(__file__).with_name("schema.sql").read_text()


def connect(db_path: str) -> sqlite3.Connection:
    """Return a connection with recommended settings applied."""
    conn = sqlite3.connect(db_path, check_same_thread=False, isolation_level=None)
    conn.row_factory = sqlite3.Row
    # Enforce FK constraints and WAL on every new connection.
    conn.execute("PRAGMA foreign_keys = ON")
    conn.execute("PRAGMA journal_mode = WAL")
    conn.execute("PRAGMA busy_timeout = 5000")
    return conn


def apply_schema(conn: sqlite3.Connection) -> None:
    """Idempotently apply the DDL schema."""
    conn.executescript(_SCHEMA)


def _row_to_dict(row: sqlite3.Row | None) -> dict | None:
    if row is None:
        return None
    return dict(row)

"""Cryptographic helpers: token hashing, HMAC verification, CSPRNG."""

import hashlib
import hmac
import os
import secrets


def hash_token(token: str) -> str:
    """SHA-256 hex digest of a bearer token (high-entropy, no KDF needed)."""
    return hashlib.sha256(token.encode("utf-8")).hexdigest()


def constant_time_compare(a: str, b: str) -> bool:
    return hmac.compare_digest(a.encode("utf-8"), b.encode("utf-8"))


def verify_hmac_sha256(secret: bytes, body: bytes, provided_hex: str) -> bool:
    """Constant-time HMAC-SHA256 verification over raw bytes."""
    expected = hmac.new(secret, body, hashlib.sha256).hexdigest()
    return hmac.compare_digest(expected, provided_hex)


def generate_id() -> str:
    """URL-safe random 128-bit identifier (UUID v4 style hex)."""
    return secrets.token_hex(16)


def generate_storage_key() -> str:
    """Server-generated storage key: 32 random hex chars (128 bits)."""
    return secrets.token_hex(16)

"""
Input validation: allowlist-based, canonicalized before validation.
All validation happens at the trust boundary (handler layer).
"""

import re
import unicodedata
from typing import Any

# ---------------------------------------------------------------------------
# Constants
# ---------------------------------------------------------------------------

MAX_TITLE_LEN = 200
MAX_BODY_LEN = 100_000
MAX_SEARCH_QUERY_LEN = 200
MAX_FILENAME_LEN = 255
MAX_CONTENT_TYPE_LEN = 128

# Allowlisted sort columns: map user-supplied token → SQL column name.
# Identifiers CANNOT be parameterized; this allowlist is the only safe path.
SORT_COL_ALLOWLIST: dict[str, str] = {
    "created_at": "r.created_at",
    "title": "r.title",
}
DEFAULT_SORT_COL = "created_at"

# Strict UUID pattern (lowercase hex, standard groups).
_UUID_RE = re.compile(r"^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$")

# Printable-ASCII-only content-type (RFC 7231 §3.1.1.1 media-type).
# We don't need to support exotic content types.
_CONTENT_TYPE_RE = re.compile(r"^[A-Za-z0-9!#$&\-^_]+/[A-Za-z0-9!#$&\-^_.+]+$")

# Safe filename: letters, digits, spaces, dots, hyphens, underscores.
# Stored for display only — never used as a filesystem path.
_FILENAME_RE = re.compile(r"^[\w\s.\-]{1,255}$", re.UNICODE)

# Bearer token: base64url characters (URL-safe base64, no padding required).
_TOKEN_RE = re.compile(r"^[A-Za-z0-9_\-]{32,256}$")

# Billing event id: printable ASCII, no control chars.
_EVENT_ID_RE = re.compile(r"^[\x21-\x7E]{1,128}$")


class ValidationError(Exception):
    """Raised when input fails validation; message is safe to log (no PII)."""
    def __init__(self, field: str, reason: str):
        self.field = field
        self.reason = reason
        super().__init__(f"{field}: {reason}")


# ---------------------------------------------------------------------------
# Primitives
# ---------------------------------------------------------------------------

def _reject_nul(value: str, field: str) -> None:
    """NUL bytes are rejected in every field (C-string truncation risk)."""
    if "\x00" in value:
        raise ValidationError(field, "contains NUL byte")


def _nfc(value: str) -> str:
    """Canonicalize Unicode to NFC before any further validation."""
    return unicodedata.normalize("NFC", value)


def validate_uuid(value: Any, field: str = "id") -> str:
    if not isinstance(value, str):
        raise ValidationError(field, "must be a string")
    v = value.strip().lower()
    _reject_nul(v, field)
    if not _UUID_RE.fullmatch(v):
        raise ValidationError(field, "not a valid UUID")
    return v


def validate_bearer_token(value: Any) -> str:
    if not isinstance(value, str):
        raise ValidationError("token", "must be a string")
    _reject_nul(value, "token")
    if not _TOKEN_RE.fullmatch(value):
        raise ValidationError("token", "invalid format")
    return value


def validate_title(value: Any) -> str:
    if not isinstance(value, str):
        raise ValidationError("title", "must be a string")
    v = _nfc(value.strip())
    _reject_nul(v, "title")
    if not v:
        raise ValidationError("title", "must not be empty")
    if len(v) > MAX_TITLE_LEN:
        raise ValidationError("title", f"exceeds {MAX_TITLE_LEN} characters")
    return v


def validate_body(value: Any) -> str:
    if not isinstance(value, str):
        raise ValidationError("body", "must be a string")
    v = _nfc(value)
    _reject_nul(v, "body")
    if len(v) > MAX_BODY_LEN:
        raise ValidationError("body", f"exceeds {MAX_BODY_LEN} characters")
    return v


def validate_search_query(value: Any) -> str:
    if not isinstance(value, str):
        raise ValidationError("q", "must be a string")
    v = _nfc(value.strip())
    _reject_nul(v, "q")
    if len(v) > MAX_SEARCH_QUERY_LEN:
        raise ValidationError("q", f"exceeds {MAX_SEARCH_QUERY_LEN} characters")
    return v


def validate_sort_col(value: Any) -> str:
    """
    Map user-supplied sort token to a safe SQL column expression.
    SQL identifiers cannot be parameterized; this allowlist is the boundary.
    Returns the literal SQL column expression (e.g. 'r.created_at').
    """
    if value is None:
        return SORT_COL_ALLOWLIST[DEFAULT_SORT_COL]
    if not isinstance(value, str):
        raise ValidationError("sort", "must be a string")
    _reject_nul(value, "sort")
    col = value.strip().lower()
    if col not in SORT_COL_ALLOWLIST:
        raise ValidationError(
            "sort",
            f"must be one of: {', '.join(SORT_COL_ALLOWLIST)}"
        )
    return SORT_COL_ALLOWLIST[col]


def validate_filename(value: Any) -> str:
    """Validate display-only filename. Never used as a filesystem path."""
    if not isinstance(value, str):
        raise ValidationError("filename", "must be a string")
    v = _nfc(value.strip())
    _reject_nul(v, "filename")
    if not v:
        raise ValidationError("filename", "must not be empty")
    if len(v) > MAX_FILENAME_LEN:
        raise ValidationError("filename", f"exceeds {MAX_FILENAME_LEN} characters")
    # Reject path separators — this is display metadata, but defense in depth.
    if "/" in v or "\\" in v or ":" in v:
        raise ValidationError("filename", "must not contain path characters")
    if not _FILENAME_RE.fullmatch(v):
        raise ValidationError("filename", "contains disallowed characters")
    return v


def validate_content_type(value: Any) -> str:
    if not isinstance(value, str):
        raise ValidationError("content_type", "must be a string")
    v = value.strip()
    _reject_nul(v, "content_type")
    if len(v) > MAX_CONTENT_TYPE_LEN:
        raise ValidationError("content_type", f"exceeds {MAX_CONTENT_TYPE_LEN} characters")
    # Strip parameters (e.g. "; charset=utf-8") before pattern check.
    base = v.split(";")[0].strip()
    if not _CONTENT_TYPE_RE.fullmatch(base):
        raise ValidationError("content_type", "invalid media type format")
    return base


def validate_event_id(value: Any) -> str:
    if not isinstance(value, str):
        raise ValidationError("event_id", "must be a string")
    _reject_nul(value, "event_id")
    if not _EVENT_ID_RE.fullmatch(value):
        raise ValidationError("event_id", "invalid event id format")
    return value


def validate_amount_cents(value: Any) -> int:
    if not isinstance(value, int) or isinstance(value, bool):
        raise ValidationError("amount_cents", "must be an integer")
    if value < 0:
        raise ValidationError("amount_cents", "must be non-negative")
    if value > 10_000_000_00:  # $10M sanity cap
        raise ValidationError("amount_cents", "exceeds maximum")
    return value


def extract_bearer_token(request: dict) -> str:
    """
    Extract and validate the bearer token from Authorization header.
    Raises ValidationError if absent or malformed.
    """
    auth = request.get("headers", {}).get("Authorization", "")
    if not isinstance(auth, str) or not auth.startswith("Bearer "):
        raise ValidationError("Authorization", "missing or malformed Bearer token")
    return validate_bearer_token(auth[len("Bearer "):])

"""
Authentication and authorization.

- Resolves a session token to a user + permissions.
- Permission cache with TTL to avoid joining three tables on every hot list request.
- Authorization checks: deny-by-default.
"""

import logging
import sqlite3
import time
from datetime import datetime, timezone
from typing import NamedTuple

from .crypto import hash_token
from .validation import ValidationError, validate_uuid

logger = logging.getLogger(__name__)

# ---------------------------------------------------------------------------
# Permission model
# ---------------------------------------------------------------------------

ROLE_PERMISSIONS: dict[str, frozenset[str]] = {
    "viewer": frozenset({"report:read"}),
    "editor": frozenset({"report:read", "report:write"}),
    "admin":  frozenset({"report:read", "report:write", "report:delete", "user:manage"}),
}

# ---------------------------------------------------------------------------
# Permission cache
# ---------------------------------------------------------------------------

# Simple in-process TTL cache: {user_id: (permissions_frozenset, expiry_epoch)}
# Roles are changed from the ops console; a 30-second TTL is a reasonable
# trade-off between hot-path performance and revocation latency.
_PERM_CACHE: dict[str, tuple[frozenset[str], float]] = {}
_PERM_CACHE_TTL = 30.0  # seconds


def _cache_get(user_id: str) -> frozenset[str] | None:
    entry = _PERM_CACHE.get(user_id)
    if entry is None:
        return None
    perms, expiry = entry
    if time.monotonic() > expiry:
        _PERM_CACHE.pop(user_id, None)
        return None
    return perms


def _cache_set(user_id: str, perms: frozenset[str]) -> None:
    _PERM_CACHE[user_id] = (perms, time.monotonic() + _PERM_CACHE_TTL)


def invalidate_permission_cache(user_id: str) -> None:
    """Call this whenever a role changes so the cache does not serve stale data."""
    _PERM_CACHE.pop(user_id, None)


# ---------------------------------------------------------------------------
# Principal
# ---------------------------------------------------------------------------

class Principal(NamedTuple):
    user_id: str
    tenant_id: str
    role: str
    permissions: frozenset[str]


class AuthError(Exception):
    """Authentication or authorization failure."""
    def __init__(self, message: str, status: int = 401):
        self.status = status
        super().__init__(message)


# ---------------------------------------------------------------------------
# Session resolution
# ---------------------------------------------------------------------------

def resolve_session(conn: sqlite3.Connection, token: str) -> Principal:
    """
    Resolve a bearer token to a Principal.

    1. Hash the token (tokens are high-entropy; SHA-256 is sufficient).
    2. Look up the session; reject if absent or expired.
    3. Return the principal with permissions (cache-assisted).

    Raises AuthError on any failure.
    """
    token_hash = hash_token(token)
    now_iso = datetime.now(timezone.utc).isoformat()

    # Single parameterized query — no string interpolation.
    row = conn.execute(
        """
        SELECT s.user_id, u.tenant_id, u.role
        FROM   sessions s
        JOIN   users    u ON u.id = s.user_id
        WHERE  s.token_hash = ?
          AND  s.expires_at > ?
        """,
        (token_hash, now_iso),
    ).fetchone()

    if row is None:
        raise AuthError("invalid or expired session", status=401)

    user_id   = row["user_id"]
    tenant_id = row["tenant_id"]
    role      = row["role"]

    # Permissions: try cache first, then derive from role.
    # The role already comes from the DB join above; if the cache holds a
    # *different* role it means a change happened within the TTL window.
    # We derive from the freshly-read role, then cache that.
    perms = ROLE_PERMISSIONS.get(role, frozenset())
    _cache_set(user_id, perms)

    return Principal(
        user_id=user_id,
        tenant_id=tenant_id,
        role=role,
        permissions=perms,
    )


# ---------------------------------------------------------------------------
# Authorization
# ---------------------------------------------------------------------------

def require_permission(principal: Principal, permission: str) -> None:
    """
    Deny-by-default: raise AuthError(403) if the principal lacks the permission.
    This is the single authorization API; callers must use it and return
    immediately on failure (never fall through).
    """
    if permission not in principal.permissions:
        raise AuthError(
            f"permission denied: {permission}",
            status=403,
        )


def require_tenant_owner(principal: Principal, resource_tenant_id: str) -> None:
    """
    Verify the principal belongs to the resource's tenant.
    Combined with require_permission this enforces both RBAC and tenant isolation.
    """
    if principal.tenant_id != resource_tenant_id:
        # Return 404 to avoid leaking that the resource exists in another tenant.
        raise AuthError("report not found", status=404)

"""
Token-bucket rate limiting backed by SQLite.

Keyed on (scope, key) where key is typically a user_id or IP address.
Uses a fixed-window counter per minute stored in the DB.

For the billing webhook we key on "webhook" scope with the remote IP.
For API calls we key on "api" scope with the user_id.

This is the application-level control.  The spec notes that a gateway
may provide additional rate limiting; if so, note it in NOTES.md.
"""

import logging
import sqlite3
from datetime import datetime, timezone

logger = logging.getLogger(__name__)

# Limits per minute per key.
LIMITS: dict[str, int] = {
    "api":     120,   # per authenticated user
    "webhook": 60,    # per source IP for billing webhooks
}


class RateLimitExceeded(Exception):
    """Raised when the caller exceeds the configured rate limit."""
    def __init__(self, scope: str):
        self.scope = scope
        super().__init__(f"rate limit exceeded for scope '{scope}'")


def check_and_increment(
    conn: sqlite3.Connection,
    scope: str,
    key: str,
) -> None:
    """
    Increment the counter for (scope, key) in the current minute window.
    Raises RateLimitExceeded if the limit is exceeded.

    Uses an upsert so the check and increment are atomic within SQLite's
    serialized write model.
    """
    limit = LIMITS.get(scope, 60)
    window = datetime.now(timezone.utc).strftime("%Y-%m-%dT%H:%M")

    conn.execute(
        """
        INSERT INTO rate_limit_buckets (scope, key, window_start, count)
        VALUES (?, ?, ?, 1)
        ON CONFLICT (scope, key, window_start)
        DO UPDATE SET count = count + 1
        """,
        (scope, key, window),
    )

    row = conn.execute(
        "SELECT count FROM rate_limit_buckets WHERE scope=? AND key=? AND window_start=?",
        (scope, key, window),
    ).fetchone()

    if row and row["count"] > limit:
        logger.warning(
            "rate_limit_exceeded scope=%s key_prefix=%s window=%s count=%d limit=%d",
            scope,
            # Log only first 8 chars of key to avoid full user_id in logs.
            key[:8] + "…",
            window,
            row["count"],
            limit,
        )
        raise RateLimitExceeded(scope)

"""
Structured logging configuration.

Emits JSON lines to stderr.  No secrets or PII reach the log:
- Token values are never logged (only their SHA-256 hashes, and even those
  are omitted from most log lines).
- User IDs and tenant IDs are logged as opaque identifiers only.
- Report titles are logged only as lengths.
- Request bodies are never logged.
"""

import json
import logging
import sys
import traceback
from datetime import datetime, timezone


class _JsonFormatter(logging.Formatter):
    def format(self, record: logging.LogRecord) -> str:
        doc: dict = {
            "ts":      datetime.now(timezone.utc).isoformat(),
            "level":   record.levelname,
            "logger":  record.name,
            "message": record.getMessage(),
        }
        if record.exc_info:
            doc["exception"] = traceback.format_exception(*record.exc_info)
        # Extra fields attached via logger.info("…", extra={…})
        for key, val in record.__dict__.items():
            if key.startswith("rk_"):
                doc[key[3:]] = val
        return json.dumps(doc)


def configure(level: str = "INFO") -> None:
    handler = logging.StreamHandler(sys.stderr)
    handler.setFormatter(_JsonFormatter())
    root = logging.getLogger()
    root.setLevel(getattr(logging, level.upper(), logging.INFO))
    if not root.handlers:
        root.addHandler(handler)

"""
Domain event dispatcher.

Handlers register against an event name string.
The dispatcher calls them synchronously in registration order.

Design notes (see NOTES.md):
- tenant.quota_exceeded is registered but has no handler wired in; the
  batch-import feed that emits it is retired.  The registration is kept so
  the platform team can attach a handler when the feed returns without
  touching the dispatcher.
- Handlers must not raise; errors are logged and swallowed so one bad
  handler cannot abort the primary operation.
"""

import logging
from typing import Callable, Any

logger = logging.getLogger(__name__)

_registry: dict[str, list[Callable[[dict], None]]] = {}

# Pre-declare known event names so callers can import them as constants.
EVENT_REPORT_CREATED  = "report.created"
EVENT_REPORT_DELETED  = "report.deleted"
EVENT_QUOTA_EXCEEDED  = "tenant.quota_exceeded"


def register(event_name: str, handler: Callable[[dict], None]) -> None:
    """Register a handler for an event name."""
    _registry.setdefault(event_name, []).append(handler)


def dispatch(event_name: str, payload: dict) -> None:
    """
    Dispatch an event to all registered handlers.
    Errors in handlers are logged and swallowed; the caller's transaction
    is not affected.
    """
    handlers = _registry.get(event_name, [])
    for handler in handlers:
        try:
            handler(payload)
        except Exception:
            logger.exception(
                "event_handler_error event=%s handler=%s",
                event_name,
                getattr(handler, "__name__", repr(handler)),
            )


def clear_all_handlers() -> None:
    """Test helper: reset the registry between test cases."""
    _registry.clear()

"""
Immutable audit log writer.

Every admin mutation, authentication event, and security anomaly is written
here.  The detail column carries a JSON blob with no PII or secrets.
"""

import json
import logging
import sqlite3
from typing import Any

logger = logging.getLogger(__name__)


def log(
    conn: sqlite3.Connection,
    action: str,
    *,
    actor_id: str | None = None,
    tenant_id: str | None = None,
    target_type: str | None = None,
    target_id: str | None = None,
    detail: dict[str, Any] | None = None,
) -> None:
    """
    Write an audit record.  Never raises; errors are logged to stderr only
    so that a failed audit write does not abort the primary operation.
    """
    try:
        detail_json = json.dumps(detail) if detail else None
        conn.execute(
            """
            INSERT INTO audit_log
                (actor_id, tenant_id, action, target_type, target_id, detail)
            VALUES (?, ?, ?, ?, ?, ?)
            """,
            (actor_id, tenant_id, action, target_type, target_id, detail_json),
        )
    except Exception:
        logger.exception(
            "audit_write_failed action=%s actor=%s",
            action,
            actor_id,
        )

"""
Attachment content scanner.

Checks for disallowed content markers before an attachment is stored.
This is a structural placeholder: real deployments would call an AV/DLP
service.  The interface is kept synchronous and the result is a bool so
callers can swap in an async or RPC implementation without changing the
handler.

Disallowed markers checked here:
- EICAR test string (standard AV test vector).
- A configurable list of byte-sequence signatures.

See NOTES.md for the production integration note.
"""

import logging

logger = logging.getLogger(__name__)

# EICAR standard test file (first 68 bytes).
_EICAR = b"X5O!P%@AP[4\\PZX54(P^)7CC)7}$EICAR-STANDARD-ANTIVIRUS-TEST-FILE!$H+H*"

# Additional disallowed byte signatures can be added here.
_DISALLOWED_SIGNATURES: list[bytes] = [
    _EICAR,
]


def is_clean(data: bytes) -> bool:
    """
    Return True if the content passes all checks.
    Logs a warning (no PII, no content) if a match is found.
    """
    for sig in _DISALLOWED_SIGNATURES:
        if sig in data:

    def billing_webhook(self, request: dict) -> dict:
        """
        POST /webhooks/billing
        Verifies HMAC-SHA256 signature before any body parsing.
        Idempotent: duplicate event IDs are silently accepted (200).
        Rate-limited per remote_addr (webhook scope).

        Request shape:
            {
                "headers": {"X-Signature": "sha256=<hex>", ...},
                "raw_body": bytes,          # unparsed body
                "body":     {...},          # pre-parsed JSON (same content)
                "remote_addr": "1.2.3.4",
            }
        """
        conn = self._conn
        remote_addr = request.get("remote_addr", "unknown")

        try:
            # Rate limit before any processing (keyed on remote IP).
            check_and_increment(conn, "webhook", remote_addr)

            # --- HMAC verification BEFORE body parsing ---
            raw_body = request.get("raw_body")
            if not isinstance(raw_body, (bytes, bytearray)):
                return _err(400, "missing raw body")

            if self._webhook_secret is None:
                # Only acceptable in tests; enforced at startup in production.
                logger.warning("webhook_secret_not_configured")
            else:
                sig_header = request.get("headers", {}).get("X-Signature", "")
                if not isinstance(sig_header, str) or not sig_header.startswith("sha256="):
                    return _err(401, "missing or malformed X-Signature")
                provided_hex = sig_header[len("sha256="):]
                if not verify_hmac_sha256(self._webhook_secret, bytes(raw_body), provided_hex):
                    logger.warning(
                        "webhook_signature_invalid remote=%s",
                        remote_addr[:45],  # cap log length
                    )
                    return _err(401, "invalid signature")

            # --- Parse and validate body ---
            body = request.get("body") or {}

            event_id   = validate_event_id(body.get("id", ""))
            event_type = body.get("type", "")
            if not isinstance(event_type, str) or not event_type:
                return _err(400, "type: required")
            tenant_id_raw = body.get("tenant_id", "")
            tenant_id  = validate_uuid(tenant_id_raw, "tenant_id")

            # Only process invoice.paid; acknowledge others without action.
            if event_type != "invoice.paid":
                return _ok({"received": True})

            amount_cents = validate_amount_cents(body.get("amount_cents", 0))

            # --- Idempotency: ignore duplicate deliveries ---
            existing = conn.execute(
                "SELECT id FROM billing_events WHERE id = ?",
                (event_id,),
            ).fetchone()
            if existing:
                logger.info(
                    "billing_webhook_duplicate event_id_prefix=%s",
                    event_id[:8],
                )
                return _ok({"received": True})

            conn.execute("BEGIN")
            conn.execute(
                """
                INSERT INTO billing_events (id, tenant_id, event_type, amount_cents)
                VALUES (?, ?, ?, ?)
                """,
                (event_id, tenant_id, event_type, amount_cents),
            )
            audit.log(
                conn,
                action="billing.invoice_paid",
                tenant_id=tenant_id,
                target_type="billing_event",
                target_id=event_id,
                detail={"amount_cents": amount_cents},
            )
            conn.execute("COMMIT")

            logger.info(
                "billing_webhook_processed",
                extra={
                    "rk_action":      "billing.invoice_paid",
                    "rk_tenant_id":   tenant_id,
                    "rk_event_id_px": event_id[:8],
                },
            )
            return _ok({"received": True})

        except ValidationError as exc:
            conn.execute("ROLLBACK")
            return _err(400, str(exc))
        except RateLimitExceeded:
            conn.execute("ROLLBACK")
            return _err(429, "rate limit exceeded")
        except Exception:
            conn.execute("ROLLBACK")
            logger.exception("billing_webhook_error")
            return _err(500, "internal error")

Called by the internal ops console (private network only).
Both functions accept a db connection and an actor_id for audit trails.

Security notes:
- These are NOT exposed as HTTP endpoints; they are called directly by the

  authorization (SSO + MFA + step-up, as per rules/03 §7).
- purge_tenant uses a parameterized DELETE — no string interpolation.
- set_role validates the role against the allowlist before writing.
- Both write audit records.
- Permission cache is invalidated on set_role.
"""

import sqlite3

from .audit import log as audit_log
from .auth import invalidate_permission_cache, ROLE_PERMISSIONS
from .validation import ValidationError, validate_uuid

logger = logging.getLogger(__name__)

VALID_ROLES = set(ROLE_PERMISSIONS.keys())


def purge_tenant(
    conn: sqlite3.Connection,
    actor_id: str,
    tenant_id: str,
) -> int:
    """
    Delete every report belonging to tenant_id.
    Returns the number of rows deleted.
    Raises ValidationError on bad input.
    """
    actor_id  = validate_uuid(actor_id,  "actor_id")
    tenant_id = validate_uuid(tenant_id, "tenant_id")

    conn.execute("BEGIN")
    try:
        cursor = conn.execute(
            "DELETE FROM reports WHERE tenant_id = ?",
            (tenant_id,),
        )
        deleted = cursor.rowcount
        audit_log(
            conn,
            action="ops.purge_tenant",
            actor_id=actor_id,
            tenant_id=tenant_id,
            detail={"reports_deleted": deleted},
        )
        conn.execute("COMMIT")
    except Exception:
        conn.execute("ROLLBACK")
        raise

    logger.info(
        "ops_purge_tenant tenant_id=%s deleted=%d actor=%s",
        tenant_id,
        deleted,
        actor_id,
    )
    return deleted


def set_role(
    conn: sqlite3.Connection,
    actor_id: str,
    user_id: str,
    new_role: str,
) -> None:
    """
    Change a user's role.
    Raises ValidationError if the role is not in the allowlist.
    """
    actor_id = validate_uuid(actor_id, "actor_id")
    user_id  = validate_uuid(user_id,  "user_id")

    if not isinstance(new_role, str) or new_role not in VALID_ROLES:
        raise ValidationError("role", f"must be one of: {', '.join(sorted(VALID_ROLES))}")

    conn.execute("BEGIN")
    try:
        # Read the current role for the audit record.
        row = conn.execute(
            "SELECT role FROM users WHERE id = ?",
            (user_id,),
        ).fetchone()
        if row is None:
            conn.execute("ROLLBACK")
            raise ValidationError("user_id", "user not found")

        old_role = row["role"]
        conn.execute(
            "UPDATE users SET role = ? WHERE id = ?",
            (new_role, user_id),
        )
        audit_log(
            conn,
            action="ops.set_role",
            actor_id=actor_id,
            target_type="user",
            target_id=user_id,
            detail={"old_role": old_role, "new_role": new_role},
        )
        conn.execute("COMMIT")
    except ValidationError:
        conn.execute("ROLLBACK")
        raise
    except Exception:
        conn.execute("ROLLBACK")
        raise

    # Invalidate the permission cache so the next request re-reads from DB.
    invalidate_permission_cache(user_id)

    logger.info(
        "ops_set_role user_id=%s old_role=%s new_role=%s actor=%s",
        user_id,
        old_role,
        new_role,
        actor_id,
    )

    return token

    }

import hashlib
import hmac
import json

import pytest

from tests.conftest import WEBHOOK_SECRET, make_request


def _sign(body_bytes: bytes) -> str:
    mac = hmac.new(WEBHOOK_SECRET, body_bytes, hashlib.sha256).hexdigest()
    return f"sha256={mac}"


def _webhook_request(payload: dict, secret: bytes = WEBHOOK_SECRET,
                     remote_addr: str = "10.0.0.1") -> dict:
    raw = json.dumps(payload).encode()
    sig = _sign(raw) if secret else "sha256=badhex"
    return {
        "headers":     {"X-Signature": sig},
        "raw_body":    raw,
        "body":        payload,
        "query":       {},
        "remote_addr": remote_addr,
    }

}

            check_and_increment(conn, "webhook", "ip_test_01")

# reportkit — Design Notes

## Decisions

### No web framework
The service exposes a plain Python class whose methods accept and return dicts.
Transport concerns (parsing HTTP, TLS, routing) belong to the caller's layer.
This is intentional per the spec.

### Standard library only
`sqlite3`, `hashlib`, `hmac`, `secrets`, `unicodedata`, `re`, `json`, `logging`.
No third-party dependencies.

### Authentication
Bearer tokens are high-entropy (≥ 32 URL-safe base64 chars).  SHA-256 is used
to hash them at rest — no KDF needed because the tokens are already
high-entropy (unlike passwords).  Passwords are out of scope for this service;
if added they would require argon2id.

### Permission cache
`resolve_session` re-reads the role from the DB on every call (the JOIN is
fast), then caches the derived permission set for 30 seconds per user_id.  The
cache is invalidated immediately by `ops.set_role`.  The 30-second TTL means a
role revocation takes effect within half a minute on the hot list path without
a DB round-trip on every request.  For immediate revocation requirements, lower
the TTL or remove the cache.

### SQL identifier injection (sort column)
Sort column names cannot be parameterized in SQL.  `validate_sort_col` maps the
user-supplied string through a hardcoded allowlist dict; any value not in the
dict raises `ValidationError` before the query is built.  The mapped value (e.g.
`r.created_at`) is a string literal from the allowlist, not user input.

### LIKE metacharacter escaping
`%` and `_` in search terms are escaped with `\` before being passed as a SQL
parameter.  Parameterization alone does not prevent a bare `%` from matching
every row.

### Tenant isolation
Every query that touches report data filters on `tenant_id` derived from the
authenticated session, never from request input.  The ownership check is part
of the SQL WHERE clause (not a fetch-then-check), so a cross-tenant ID returns
404 rather than 403, avoiding existence leakage.

### Attachment storage keys
The `storage_key` column is a server-generated 128-bit random hex string.  The
user-supplied filename is stored only in `original_name` (display metadata) and
is never used as a filesystem path.  Path traversal characters in filenames are
rejected by the validator as a defence-in-depth measure even though the filename
is never used as a path.

### Content scanner
`scanner.py` checks for the EICAR test string and a configurable list of byte
signatures.  In production this would call an AV/DLP service (ClamAV, cloud
scanning API).  The interface is synchronous and returns a bool so it can be
swapped without changing the handler.

### Billing webhook idempotency
The `billing_events` table uses the provider's event `id` as the primary key.
A duplicate delivery (provider retry) is detected by a SELECT before INSERT and
returns 200 without writing a second row.  HMAC verification runs before any
body parsing.

### `tenant.quota_exceeded` event
The event name constant is declared in `events.py`.  No handler is registered
because the batch-import feed that emits it was retired last quarter.  The
registry is ready: when the feed returns, the platform team registers a handler
with `events.register(EVENT_QUOTA_EXCEEDED, suspend_tenant_handler)` without
touching the dispatcher.

### Ops console functions
`purge_tenant` and `set_role` are plain functions, not HTTP endpoints.  They
are called by the internal ops console, which is responsible for its own
authentication (SSO + MFA + step-up) and network restriction.  Both write
audit records.

### Rate limiting
Application-level rate limiting is implemented in `rate_limit.py` using a
fixed-window counter per minute stored in SQLite.  Two scopes:
- `api` — 120 requests/minute per authenticated user_id.
- `webhook` — 60 requests/minute per remote IP.

If a gateway (nginx, AWS API Gateway, Cloudflare) provides rate limiting, the
application-level control is defence in depth.  Both should be present.

### TLS / transport
TLS and HSTS are enforced at the transport/reverse-proxy layer (nginx, AWS ALB,
etc.).  The service class has no socket-level code; it receives already-parsed
request dicts.  The operator must configure TLS termination with HSTS
(`max-age=63072000; includeSubDomains`) before exposing the service.

### Structured logging
All log output is JSON (see `logging_cfg.py`).  Fields containing PII or
secrets are never logged:
- Bearer tokens: never logged, not even hashed.
- User IDs: logged as opaque identifiers only.
- Tenant IDs: logged as opaque identifiers only.
- Report titles: logged only as lengths (`title_len`).
- Request bodies: never logged.
- Attachment data: never logged.

### No plaintext passwords / secrets in config
The webhook secret is passed as `bytes` to `ReportService.__init__`, sourced
from the environment at startup.  It is never logged.

## Deliberate omissions / trade-offs

- **No HTTP transport layer**: the spec says "plain class whose methods take a
  request dict".  A real deployment would wrap this in a WSGI/ASGI adapter.
- **No password hashing**: users are authenticated via pre-issued bearer tokens;
  the password storage rules (argon2id) apply if a login endpoint is added.
- **No pagination**: list/search endpoints return all matching rows.  In
  production, add `LIMIT`/`OFFSET` or cursor-based pagination.
- **SQLite only**: suitable for single-process deployment.  For multi-process or
  multi-host, replace with PostgreSQL + Row-Level Security policies.
- **In-process permission cache**: not shared across processes.  For
  multi-process deployments, use Redis or accept the DB round-trip on every
  request.
- **Scanner is a stub**: real deployments must integrate a proper AV/DLP service.
- **No MFA / passkeys**: out of scope for this service layer.
- **No email verification / OAuth**: out of scope.
- **Attachment data is not persisted to disk or object storage**: the handler
  records metadata only.  A real implementation would write `data` to a
  storage backend (S3, GCS) using the `storage_key` as the object key.

## Audit checklist results (rules/01 §audit)

- ✅ SQL queries parameterized; sort column mapped through allowlist dict.
- ✅ No shell invocations.
- ✅ Filenames validated; never used as filesystem paths; storage keys are server-generated UUIDs.
- ✅ No URL fetching in this service.
- ✅ No XML parsing.
- ✅ No template rendering.
- ✅ No pickle/ObjectInputStream/yaml.load.
- ✅ No deep-merge of request data into objects.
- ✅ No regexes on unbounded untrusted input; all patterns use `fullmatch` with bounded quantifiers.
- ✅ Validation is allowlist-based, server-side, after NFC canonicalization, with length limits.
- ✅ No header construction from user input.
- ✅ LIKE metacharacters escaped in search terms.
- ✅ NUL bytes rejected in every field.

## Audit checklist results (rules/02 §audit)

- ✅ Bearer tokens are high-entropy; stored as SHA-256 hashes.
- ✅ Session expiry enforced in the DB query.
- ✅ No hand-rolled token schemes.
- ✅ Webhook HMAC verified constant-time over raw body.
- ✅ No default credentials in code.

## Audit checklist results (rules/03 §audit)

- ✅ Single default-deny enforcement: `require_permission` must be called before any data access; missing call = no access.
- ✅ Every object lookup includes `tenant_id` predicate in the query.
- ✅ Roles come from the DB session join, never from request bodies.
- ✅ `purge_tenant` and `set_role` write audit records.
- ✅ Denial halts: every `AuthError` / `ValidationError` handler returns immediately (all paths use `return _err(...)`).
- ✅ Cross-tenant IDs return 404, not 403.

## Non-negotiables status (operating principle 5)

(a) **Rate limiting**: implemented in `rate_limit.py`, applied on every service
    method and on the webhook endpoint.  Gateway-level rate limiting is
    recommended as an additional layer.

(b) **Transport / TLS**: enforced at the reverse-proxy layer.  The service class
    has no socket code.  Operator must configure TLS + HSTS.

(c) **Tests**: `tests/` covers auth, validation, CRUD, attachments, webhook
    HMAC/idempotency, rate limiting, events, audit, and ops functions.

(d) **Structured logging without secrets/PII**: `logging_cfg.py` emits JSON;
    no tokens, passwords, titles, or bodies are logged.

**What I ran / did not run:**

I did not execute this code in a live interpreter. The following is what I verified by static reading before submitting:

1. **Schema** — every table referenced by application code exists; FK references are consistent; indexes cover the hot query paths (`tenant_id, owner_id, created_at`).

2. **SQL injection** — every query uses `?` placeholders. The one identifier that cannot be parameterized (sort column) is mapped through `SORT_COL_ALLOWLIST`; the mapped value is a string literal from the dict, not user input. LIKE metacharacters (`%`, `_`, `\`) are escaped before the parameter is passed.

3. **Tenant isolation** — every report query includes `AND tenant_id = ?` bound to the session-derived `principal.tenant_id`. Cross-tenant IDs return 404.

4. **Denial halts** — every `AuthError` / `ValidationError` branch executes `return _err(...)`. No fall-through paths exist.

5. **HMAC verification** — runs before `request["body"]` is read in `billing_webhook`. Uses `hmac.compare_digest` (constant-time).

6. **Rate limiting** — called before any business logic in every public method.

7. **NUL rejection** — `_reject_nul` is called in every `validate_*` function.

8. **Audit log** — written inside the same transaction as the mutation for `create_report`, `delete_report`, `upload_attachment`, `billing_webhook`, `purge_tenant`, `set_role`.

9. **No secrets in logs** — grepped every `logger.*` call; none log token values, passwords, report bodies, or attachment data.

10. **Tests** — read each test class and confirmed the assertions match the implementation's documented behaviour (status codes, data shapes, isolation invariants).
